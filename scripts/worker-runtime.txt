const CACHE=new Map(),RATE=new Map();
function publicUrl(raw){const u=new URL(raw),host=u.hostname.toLowerCase();if(!['https:','http:'].includes(u.protocol)||u.username||u.password||u.port&&!['80','443'].includes(u.port)||!host.includes('.')||/^(?:\d|\[)|(?:^|\.)(?:localhost|local|internal|test|invalid)$/.test(host))throw Error('公開されたPDF URLを指定してください');return u;}
async function remote(raw,pdf=false,asset=false){
 let current=raw;
 for(let hop=0;hop<6;hop++){
  const u=publicUrl(current),r=await fetch(u.href,{redirect:'manual',signal:AbortSignal.timeout(40000),headers:{'User-Agent':'TDnetEarningsCompanion/14','Accept':pdf?'application/pdf':'text/html,application/json,text/javascript','Accept-Language':'ja,en;q=0.8'}});
  if([301,302,303,307,308].includes(r.status)){const loc=r.headers.get('location');if(!loc)throw Error('転送先がありません');current=new URL(loc,u).href;continue;}
  if(!r.ok)throw Error(`取得先がHTTP ${r.status}を返しました`);
  const limit=(pdf||asset?40:8)*1024*1024;if(Number(r.headers.get('content-length'))>limit)throw Error('ファイルサイズ超過');
  const chunks=[],reader=r.body.getReader();let size=0;
  try{for(;;){const {done,value}=await reader.read();if(done)break;size+=value.length;if(size>limit)throw Error('ファイルサイズ超過');chunks.push(value);}}catch(e){await reader.cancel();throw e;}
  const bytes=new Uint8Array(size);let offset=0;for(const chunk of chunks){bytes.set(chunk,offset);offset+=chunk.length;}
  if(pdf||asset){if(pdf&&(new TextDecoder().decode(bytes.slice(0,5))!=='%PDF-'||!new TextDecoder().decode(bytes.slice(-8192)).includes('%%EOF')))throw Error('生PDFの先頭・末尾を確認できません。取得が途中で切れた可能性があります');const length=Number(r.headers.get('content-length'));if(length&&!r.headers.get('content-encoding')&&length!==bytes.length)throw Error('取得サイズが一致しません');return bytes;}
  const charset=/charset=([^;\s]+)/i.exec(r.headers.get('content-type')||'')?.[1]||'utf-8';return new TextDecoder(charset).decode(bytes);
 }
 throw Error('転送回数超過');
}
